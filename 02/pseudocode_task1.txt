input_file = openRead("input.txt")
safe_report_count = 0

// Report checking function that takes a sequence of integers
function check_report_safe(report)
    previous_value = report[0]
    current_value = report[1]
    sign = signum(current_value - previous_value)

    for i = 2 to report.length
        previous_value = report[i - 1]
        current_value = report[i]
        difference = current_value - previous_value

        if abs(difference) < 1
            OR abs(difference) > 3
            OR signum(difference) != sign then
            return 0
    next i

    return 1
endfunction

// Read input file; split each line into sequence of integers and check if safe
while NOT input_file.endOfFile()
    line = input_file.readLine()
    report = line.split(" ")
    safe_report_count += check_report_safe(report);
endwhile

// Display result
print(safe_report_count)

input_file = openRead("input.txt")
array left[1000]
array right[1000]
line_number = 0
distance_sum = 0

// Read input file; parse each line and put numbers into two arrays
while NOT input_file.endOfFile()
    line = input_file.readLine()
    left[line_number], right[line_number] = line.parseFormat("{int}   {int}")
    line_number++
endwhile

// Sort arrays
sort(left)
sort(right)

// Loop through arrays; calculate distance between values in each pair and sum them
for i = 0 to 1000
    distance = abs(left[i] - right[i])
    distance_sum += distance
next i

// Display result
print(distance_sum)
